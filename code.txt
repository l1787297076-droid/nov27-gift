    function sampleTextPoints(text, maxPoints, fontScale = 1.0) {
      const w = Math.max(300, Math.floor(state.width));
      const h = Math.max(300, Math.floor(state.height));
      textSampleCanvas.width = w;
      textSampleCanvas.height = h;
      textSampleCtx.clearRect(0, 0, w, h);

      const isMobile = w < 600;
      const charCount = text.length;
      let fontSize = isMobile
        ? Math.min(w * 0.82 / Math.max(charCount * 0.82, 1.4), 124)
        : Math.min(w * 0.65 / Math.max(charCount * 0.85, 1.5), 165);
      fontSize = Math.floor(fontSize * fontScale);

      textSampleCtx.fillStyle = '#FFFFFF';
      textSampleCtx.textAlign = 'center';
      textSampleCtx.textBaseline = 'middle';
      textSampleCtx.font = `700 ${fontSize}px "Noto Serif SC", "PingFang SC", sans-serif`;
      textSampleCtx.fillText(text, w * 0.5, h * 0.45);

      const imgData = textSampleCtx.getImageData(0, 0, w, h).data;
      const step = isMobile ? 3 : 4;
      const rawPoints = [];

      for (let y = 0; y < h; y += step) {
        for (let x = 0; x < w; x += step) {
          const alpha = imgData[(y * w + x) * 4 + 3];
          if (alpha > 128) {
            rawPoints.push({ x, y });
          }
        }
      }

      if (rawPoints.length === 0) return [];

      for (let i = rawPoints.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        const tmp = rawPoints[i];
        rawPoints[i] = rawPoints[j];
        rawPoints[j] = tmp;
      }

      const selected = [];
      for (let i = 0; i < maxPoints; i++) {
        const src = rawPoints[i % rawPoints.length];
        selected.push({
          x: src.x + (Math.random() - 0.5) * 2.2,
          y: src.y + (Math.random() - 0.5) * 2.2
        });
      }
      return selected;
    }

    function assignCakeTargets() {
      const cx = state.width * 0.5;
      const isPortrait = state.height >= state.width;
      const cakeWidth = Math.min(state.width * (isPortrait ? 0.68 : 0.38), 310);
      const baseY = state.height * 0.68;

      const tiers = [
        { rx: cakeWidth * 0.5, ry: cakeWidth * 0.14, h: cakeWidth * 0.22, yBottom: baseY },
        { rx: cakeWidth * 0.36, ry: cakeWidth * 0.11, h: cakeWidth * 0.20, yBottom: baseY - cakeWidth * 0.22 },
        { rx: cakeWidth * 0.23, ry: cakeWidth * 0.08, h: cakeWidth * 0.18, yBottom: baseY - cakeWidth * 0.42 }
      ];

      const topTierTopY = tiers[2].yBottom - tiers[2].h;
      const candleHeight = cakeWidth * 0.34;
      const candleRadius = Math.max(5, cakeWidth * 0.024);

      cakeGeometry = {
        cx,
        baseY,
        candleTopX: cx,
        candleTopY: topTierTopY - candleHeight,
        candleBottomY: topTierTopY,
        scriptY: baseY + cakeWidth * 0.23,
        scaleRef: cakeWidth
      };

      const cakeParticlesCount = Math.floor(particles.length * 0.78);
      const candleParticlesCount = Math.floor(particles.length * 0.10);

      for (let i = 0; i < particles.length; i++) {
        const p = particles[i];
        if (i < cakeParticlesCount) {
          const tierIdx = i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2;
          const t = tiers[tierIdx];
          const angle = Math.random() * Math.PI * 2;
          const isTopSurface = Math.random() < 0.34;
          const radFactor = isTopSurface ? Math.sqrt(Math.random()) : 0.94 + Math.random() * 0.06;
          const yOffset = isTopSurface ? t.h : Math.random() * t.h;

          const x3d = Math.cos(angle) * t.rx * radFactor;
          const z3d = Math.sin(angle) * t.rx * radFactor;
          const y3d = t.yBottom - yOffset + Math.sin(angle) * t.ry * radFactor;

          p.tx = cx + x3d;
          p.ty = y3d;
          p.tz = z3d;
          p.role = 'cake';
          if (Math.abs(yOffset - t.h) < 4) {
            p.color = '#FFFDF9';
          } else {
            p.color = PASTEL_COLORS[(tierIdx + (i % 4)) % PASTEL_COLORS.length];
          }
          p.alpha = z3d >= -15 ? 0.92 : 0.48;
        } else if (i < cakeParticlesCount + candleParticlesCount) {
          const angle = Math.random() * Math.PI * 2;
          const hFrac = Math.random();
          const r = candleRadius * (0.3 + Math.random() * 0.7);
          p.tx = cx + Math.cos(angle) * r;
          p.ty = topTierTopY - hFrac * candleHeight;
          p.tz = Math.sin(angle) * r;
          p.role = 'candle';
          p.color = hFrac > 0.85 ? '#FFFDF9' : '#FFB7C5';
          p.alpha = 0.96;
        } else {
          p.tx = Math.random() * state.width;
          p.ty = Math.random() * state.height;
          p.tz = 0;
          p.role = 'ambient';
          p.color = PASTEL_COLORS[i % PASTEL_COLORS.length];
          p.alpha = 0.36;
        }
      }
    }

    function createIllustratedStarlightCanvas(photo, index) {
      const w = 360;
      const h = 450;

      c.width = w;
      c.height = h;

      orbGrad.addColorStop(0, '#FFFDF9');
      orbGrad.addColorStop(0.42, photo.colors[3] + 'AA');
      orbGrad.addColorStop(1, 'rgba(255,253,249,0)');
      g.fillStyle = orbGrad;
      g.beginPath();
      g.arc(orbX, orbY, 68, 0, Math.PI * 2);
      g.fill();

      g.fillStyle = '#FFFDF9';
      for (let s = 0; s < 28; s++) {
        const sx = 18 + ((index * 53 + s * 97) % (w - 36));
        const sy = 18 + ((index * 29 + s * 61) % Math.floor(h * 0.62));
        g.globalAlpha = 0.45 + (s % 5) * 0.1;
        g.beginPath();
        g.arc(sx, sy, (s % 2) + 1.1, 0, Math.PI * 2);
        g.fill();
      }
      g.globalAlpha = 1;

      g.fillStyle = 'rgba(18, 8, 38, 0.56)';
      g.beginPath();
      g.moveTo(0, h);
      g.lineTo(0, h * 0.66);
      g.quadraticCurveTo(w * 0.32, h * (0.45 + (index % 3) * 0.04), w * 0.62, h * 0.63);
      g.quadraticCurveTo(w * 0.84, h * 0.53, w, h * 0.68);
      g.lineTo(w, h);
      g.closePath();
      g.fill();

      const waterGrad = g.createLinearGradient(0, h * 0.70, 0, h);
      waterGrad.addColorStop(0, 'rgba(255, 183, 197, 0.26)');
      waterGrad.addColorStop(1, 'rgba(15, 6, 32, 0.84)');

    function createPhotoRenderedCanvas(img) {
      const w = 360;
      const h = 450;
      const pCanvas = document.createElement('canvas');
      pCanvas.width = w;
      pCanvas.height = h;
      const pctx = pCanvas.getContext('2d');

      const srcAspect = img.width / img.height;
      const dstAspect = w / h;
      let sx = 0, sy = 0, sw = img.width, sh = img.height;
      if (srcAspect > dstAspect) {
        sw = img.height * dstAspect;
        sx = (img.width - sw) * 0.5;
      } else {
        sh = img.width / dstAspect;
        sy = (img.height - sh) * 0.5;
      }
      pctx.drawImage(img, sx, sy, sw, sh, 0, 0, w, h);

      drawHolographicFrameOverlay(pctx, w, h);
      return pCanvas;
    }

    function loadExternalScript(src, timeoutMs = 4000) {
      return new Promise((resolve, reject) => {
        const script = document.createElement('script');
        script.src = src;
        script.async = true;
        script.crossOrigin = 'anonymous';
        let settled = false;
        const timer = setTimeout(() => {
          if (!settled) {
            settled = true;
            reject(new Error('Script load timeout'));
          }
        }, timeoutMs);
        script.onload = () => {
          if (!settled) {
            settled = true;
            clearTimeout(timer);
            resolve();
          }
        };
        script.onerror = () => {
          if (!settled) {
            settled = true;
            clearTimeout(timer);
            reject(new Error('Script load error'));
          }
        };
        document.head.appendChild(script);
      });
    }

     */
    function dist2D(a, b) {
      return Math.hypot(a.x - b.x, a.y - b.y);
    }

    function classifyHandLandmarks(lm, handednessLabel) {
      const wrist = lm[0];
      const middleMcp = lm[9];
      const palmSize = Math.max(0.02, dist2D(wrist, middleMcp));

      const indexOpen = dist2D(lm[8], wrist) > dist2D(lm[6], wrist) * 1.13 && dist2D(lm[8], lm[5]) > dist2D(lm[6], lm[5]) * 1.12;
      const middleOpen = dist2D(lm[12], wrist) > dist2D(lm[10], wrist) * 1.13 && dist2D(lm[12], lm[9]) > dist2D(lm[10], lm[9]) * 1.12;
      const ringOpen = dist2D(lm[16], wrist) > dist2D(lm[14], wrist) * 1.13 && dist2D(lm[16], lm[13]) > dist2D(lm[14], lm[13]) * 1.12;
      const pinkyOpen = dist2D(lm[20], wrist) > dist2D(lm[18], wrist) * 1.13 && dist2D(lm[20], lm[17]) > dist2D(lm[18], lm[17]) * 1.12;
      const thumbOpen = dist2D(lm[4], lm[17]) > dist2D(lm[3], lm[17]) * 1.16 && dist2D(lm[4], lm[5]) > palmSize * 0.50;

      const count = (thumbOpen ? 1 : 0) + (indexOpen ? 1 : 0) + (middleOpen ? 1 : 0) + (ringOpen ? 1 : 0) + (pinkyOpen ? 1 : 0);

      const isFist = count === 0 && !indexOpen && !middleOpen && !ringOpen && !pinkyOpen;
      const pinchDist = dist2D(lm[4], lm[8]);
      const isPinch = pinchDist < palmSize * 0.42 && !(middleOpen && ringOpen && pinkyOpen);
      const isThumbsUp = thumbOpen && lm[4].y < lm[3].y && lm[4].y < lm[5].y && !indexOpen && !middleOpen && !ringOpen;

      const isRight = handednessLabel === 'Right';
      const isRightOpen = isRight && count === 5;
      const isPeace = isRight && indexOpen && middleOpen && !ringOpen && !pinkyOpen && !isPinch;
      const isOpenFive = count === 5 && indexOpen && middleOpen && ringOpen && pinkyOpen && thumbOpen;

      // Mutually-exclusive Scene 6 holographic gesture priority
      let s6Gesture = 'GLIDE';
      if (isPeace) {
        s6Gesture = 'PEACE';
      } else if (isFist || isPinch) {
        s6Gesture = 'GRAB';
      } else if (isOpenFive) {
        s6Gesture = 'OPEN_PALM';
      }

      return {
        count,
        palmX: lm[9].x,
        palmY: lm[9].y,
        isFist,
        isThumbsUp,
        isRightOpen,
        s6Gesture
      };
    }

    function onHandsResults(results) {
      state.visionBusy = false;
      const landmarksList = results.multiHandLandmarks || [];
      const handednessList = results.multiHandedness || [];

      if (landmarksList.length === 0) {
        state.handDetected = false;
        state.isFist = false;
        state.isThumbsUp = false;
        state.isRightOpen = false;
        state.s6RawGesture = 'NONE';
        state.lastWavePalmX = null;
        return;
      }

      state.handDetected = true;

      let primary = null;
      let anyFist = false;
      let anyThumbsUp = false;
      let anyRightOpen = false;
      let chosenS6Gesture = 'GLIDE';

      for (let i = 0; i < landmarksList.length; i++) {
        const label = handednessList[i] ? handednessList[i].label : 'Right';
        const info = classifyHandLandmarks(landmarksList[i], label);
        if (i === 0) primary = info;
        if (info.isFist) anyFist = true;
        if (info.isThumbsUp) anyThumbsUp = true;
        if (info.isRightOpen) anyRightOpen = true;
        if (info.s6Gesture !== 'GLIDE') {
          chosenS6Gesture = info.s6Gesture;
        }
      }

      state.fingerCount = primary.count;
      state.palmX = primary.palmX;
      state.palmY = primary.palmY;
      state.isFist = anyFist;
      state.isThumbsUp = anyThumbsUp;
      state.isRightOpen = anyRightOpen;
      state.s6RawGesture = chosenS6Gesture;

      if (state.lastWavePalmX !== null) {
        const dx = Math.abs(primary.palmX - state.lastWavePalmX);
        if (dx > 0.015 && dx < 0.35) {
          state.waveEnergy += dx;
        }
      }
      state.lastWavePalmX = primary.palmX;
    }

    function onFaceMeshResults(results) {
      state.visionBusy = false;
      const faces = results.multiFaceLandmarks || [];
      if (faces.length === 0) {
        state.faceDetected = false;
        return;
      }

      state.faceDetected = true;
      state.lastFaceSeenTime = performance.now();
      const lm = faces[0];

      const outerEyeSpan = dist2D(lm[33], lm[263]);
      const faceWidth = dist2D(lm[234], lm[454]);
      const faceSize = Math.max(outerEyeSpan, faceWidth, 0.04);

      const mouthWidth = dist2D(lm[61], lm[291]);
      const innerLipHeight = dist2D(lm[13], lm[14]);

      const normMouthWidth = mouthWidth / faceSize;
      const normInnerHeight = innerLipHeight / faceSize;

      // 0.5s personal baseline calibration (median)
      if (state.s4Calibrating) {
        state.s4CalibSamples.push({ faceSize, normMouthWidth });
        if (performance.now() - state.s4CalibStartTime >= 500 && state.s4CalibSamples.length >= 4) {
          const sortedSizes = state.s4CalibSamples.map(s => s.faceSize).sort((a, b) => a - b);
          const sortedMouths = state.s4CalibSamples.map(s => s.normMouthWidth).sort((a, b) => a - b);
          const mid = Math.floor(sortedSizes.length / 2);
          state.baseFaceSize = sortedSizes[mid];
          state.baseNormMouthWidth = sortedMouths[mid];
          state.s4Calibrating = false;
        }
        return;
      }

      // Dual-channel blow detection
      if (state.scene === 4 && state.candleLit && !state.candleExtinguishedLocked) {
        const faceSizeRatio = faceSize / state.baseFaceSize;
        const mouthWidthRatio = normMouthWidth / state.baseNormMouthWidth;

        // Dynamic flame shrink & tilt while approaching (1.04 -> 1.25)
        if (faceSizeRatio > 1.04) {
          const leanProgress = Math.min(1, (faceSizeRatio - 1.04) / (1.25 - 1.04));
          state.flameTilt = leanProgress * 0.88;
          state.flameScaleY = Math.max(0.42, 1 - leanProgress * 0.55);
        } else {
          state.flameTilt *= 0.85;
          state.flameScaleY += (1 - state.flameScaleY) * 0.15;
        }

        // Channel 1: Pursed lips (not wide open, includes lenient path)
        const notWideOpen = normInnerHeight < 0.145 && mouthWidthRatio < 0.92;
        const pursedStrict = notWideOpen && mouthWidthRatio <= 0.81 && normInnerHeight >= 0.016;
        const pursedLenient = notWideOpen && mouthWidthRatio <= 0.77;

        // Channel 2: Approach / lean-in (faceSizeRatio >= 1.25)
        const approachedClose = faceSizeRatio >= 1.25;

        if (pursedStrict || pursedLenient || approachedClose) {
          extinguishCandle();
        }
      }
    }

    function stepVisionPipeline(now) {
      if (!state.cameraGranted || videoEl.readyState < 2) return;

      if (camPreviewCanvas.width !== 96) {
        camPreviewCanvas.width = 96;
        camPreviewCanvas.height = 120;
      }
      camPreviewCtx.save();
      camPreviewCtx.translate(96, 0);
      camPreviewCtx.scale(-1, 1);
      camPreviewCtx.drawImage(videoEl, 0, 0, 96, 120);
      camPreviewCtx.restore();

      if (state.visionBusy || now - state.lastVisionRun < 44) return;
      state.lastVisionRun = now;

      if (state.scene === 4) {
        if (state.faceReady && mpFaceMesh) {
          state.visionBusy = true;
          mpFaceMesh.send({ image: videoEl }).catch(() => { state.visionBusy = false; });
        }
      } else if ([1, 2, 3, 5, 6].includes(state.scene)) {
        if (state.handsReady && mpHands) {
          state.visionBusy = true;
          mpHands.send({ image: videoEl }).catch(() => { state.visionBusy = false; });
        }
      }
    }

    /**
     * ============================================================================
     * MAIN FRAME UPDATE & RENDER LOOP
     * ============================================================================
     */
    function updateAndRender(now) {
      requestAnimationFrame(updateAndRender);

      const rawDt = (now - state.lastFrameTime) / 1000;
      state.lastFrameTime = now;
      state.dt = Math.min(rawDt, 0.05);
      state.time += state.dt;

      // Monitor real-time FPS and adapt particle budget if FPS < 38
      if (rawDt > 0) {
        state.fpsHistory.push(1 / rawDt);
        if (now - state.lastFpsCheck > 1200 && state.fpsHistory.length >= 20) {
          const avgFps = state.fpsHistory.reduce((a, b) => a + b, 0) / state.fpsHistory.length;
          state.fpsHistory.length = 0;
          state.lastFpsCheck = now;

          if (avgFps < 38 && state.qualityTier > 0) {
            state.qualityTier -= 1;
            const reducedCount = Math.max(1300, Math.floor(particles.length * 0.78));
            particles.length = reducedCount;
          }
        }
      }

      stepVisionPipeline(now);

      state.smoothPalmX += (state.palmX - state.smoothPalmX) * 0.22;
      state.smoothPalmY += (state.palmY - state.smoothPalmY) * 0.22;

      ctx.clearRect(0, 0, state.width, state.height);

      // 1. Deep purple starry sky background stars, nebula glow & shooting stars
      renderBackgroundStars();

      // 2. Scene-specific logic & foreground rendering
      if (state.scene === 1) {
        updateAndRenderScene1(now);
      } else if (state.scene === 2) {
        updateAndRenderScene2();
      } else if (state.scene === 3 || state.scene === 4) {
        updateAndRenderScene3And4();
      } else if (state.scene === 5) {
        updateAndRenderScene5();
      } else if (state.scene === 6) {
        updateAndRenderScene6();
      }

      // 3. Main adaptive particle cloud
      updateAndRenderParticles();

      // 4. Transient FX particles
      updateAndRenderFxParticles();

      // 5. Scene 6 Holographic Palm Stardust Cursor Overlay
      if (state.scene === 6 && state.handDetected && !state.s6ExitingVortex) {
        renderScene6HoloCursor();
      }

    function renderBackgroundStars() {
      // Subtle romantic nebula glow
      const neb1 = ctx.createRadialGradient(state.width * 0.28, state.height * 0.28, 10, state.width * 0.28, state.height * 0.28, state.width * 0.55);
      neb1.addColorStop(0, 'rgba(255, 183, 197, 0.055)');
      neb1.addColorStop(1, 'rgba(9, 4, 20, 0)');
      ctx.fillStyle = neb1;
      ctx.fillRect(0, 0, state.width, state.height);

      const neb2 = ctx.createRadialGradient(state.width * 0.75, state.height * 0.65, 10, state.width * 0.75, state.height * 0.65, state.width * 0.52);
      neb2.addColorStop(0, 'rgba(216, 180, 254, 0.05)');
      neb2.addColorStop(1, 'rgba(9, 4, 20, 0)');
      ctx.fillStyle = neb2;
      ctx.fillRect(0, 0, state.width, state.height);

      // Multi-layer twinkling stars
      for (let i = 0; i < bgStars.length; i++) {
        const s = bgStars[i];
        const twinkle = 0.35 + 0.65 * (0.5 + 0.5 * Math.sin(state.time * s.speed + s.phase));
        ctx.globalAlpha = twinkle * 0.78;
        ctx.fillStyle = s.color;
        ctx.beginPath();
        ctx.arc(s.x * state.width, s.y * state.height, s.r, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;

      // Occasional gentle shooting stars
      if (state.time >= nextShootingStarTime) {
        spawnShootingStar();
        nextShootingStarTime = state.time + 4.5 + Math.random() * 5.0;
      }

      for (let i = shootingStars.length - 1; i >= 0; i--) {
        const ss = shootingStars[i];
        ss.x += ss.vx;
        ss.y += ss.vy;
        ss.life -= ss.decay;
        if (ss.life <= 0) {
          shootingStars.splice(i, 1);
          continue;
        }
        const norm = Math.hypot(ss.vx, ss.vy) || 1;
        const tx = ss.x - (ss.vx / norm) * ss.len * ss.life;
        const ty = ss.y - (ss.vy / norm) * ss.len * ss.life;
        const grad = ctx.createLinearGradient(ss.x, ss.y, tx, ty);
        grad.addColorStop(0, '#FFFDF9');
        grad.addColorStop(0.4, ss.color + '88');
        grad.addColorStop(1, 'rgba(255, 253, 249, 0)');
        ctx.save();
        ctx.globalAlpha = ss.life * 0.75;
        ctx.strokeStyle = grad;
        ctx.lineWidth = 1.6;
        ctx.lineCap = 'round';
        ctx.beginPath();
        ctx.moveTo(ss.x, ss.y);
        ctx.lineTo(tx, ty);
        ctx.stroke();
        ctx.restore();
      }
    }

    /**

      if (!state.s6ExitingVortex) {
        if (!state.handDetected) {
          state.s6RawGesture = 'NONE';
          state.s6StableGesture = 'NONE';
          state.s6GestureDebounceTimer = 0;

          if (state.s6GrabbedIdx !== -1) {
            effectiveRaw = effectiveRaw === 'OPEN_PALM' ? 'OPEN_PALM' : 'HOLDING_VIEW';
          } else if (effectiveRaw === 'OPEN_PALM') {
            effectiveRaw = 'GLIDE';
          }

          if (effectiveRaw !== state.s6StableGesture) {
            state.s6GestureDebounceTimer += state.dt;
            if (state.s6GestureDebounceTimer >= 0.14) {
              state.s6StableGesture = effectiveRaw;
              state.s6GestureDebounceTimer = 0;
              state.s6ChargeTimer = 0;
            }
          } else {
            state.s6GestureDebounceTimer = 0;
          }

          if (state.s6StableGesture === 'GLIDE' && state.s6GrabbedIdx === -1) {
            state.s6ChargeTimer = 0;
            const offset = state.smoothPalmX - 0.5;
            const deadzone = 0.085;
            if (Math.abs(offset) > deadzone) {
              const norm = (Math.abs(offset) - deadzone) / (0.5 - deadzone);
              const speed = Math.sign(offset) * Math.pow(norm, 1.35) * 4.4;
              state.ribbonVel = speed * state.dt;
              state.ribbonPos += state.ribbonVel;
            } else {
              const nearest = Math.round(state.ribbonPos);
              state.ribbonPos += (nearest - state.ribbonPos) * 0.14;
              state.ribbonVel = 0;
            }
          } else if (state.s6StableGesture === 'GRAB' && state.s6GrabbedIdx === -1) {
            const nearest = Math.round(state.ribbonPos);
            state.ribbonPos += (nearest - state.ribbonPos) * 0.22;

              triggerAirGrabPhoto(getFrontRibbonIndex());

            state.s6ChargeTimer += state.dt;
            if (state.s6ChargeTimer >= 0.35) {
              state.s6ChargeTimer = 0;
              triggerReleasePhoto();
            }
          } else if (state.s6StableGesture === 'PEACE' && state.s6GrabbedIdx === -1) {
            state.s6ChargeTimer += state.dt;
            if (state.s6ChargeTimer >= 0.50) {
              state.s6ChargeTimer = 0;
              triggerVortexExitToFinale();
            }
          }

          state.s6CursorTrailTimer += state.dt;
          if (state.s6CursorTrailTimer >= 0.03) {

              x: state.smoothPalmX * state.width + (Math.random() - 0.5) * 10,

              size: 2.2 + Math.random() * 1.4,
              color: state.s6StableGesture === 'PEACE' ? '#FFB7C5' : '#F7E3AF',
              type: 'spark'
            });
          }
        }
      }

      if (!state.isDraggingRibbon && (!state.handDetected || state.s6StableGesture !== 'GLIDE') && state.s6GrabbedIdx === -1) {
        state.ribbonPos += state.ribbonVel;
        state.ribbonVel *= 0.90;
        if (Math.abs(state.ribbonVel) < 0.008) {
          const nearest = Math.round(state.ribbonPos);
          state.ribbonPos += (nearest - state.ribbonPos) * 0.14;
        }
      }

      const targetEnlarge = state.s6GrabbedIdx !== -1 ? 1 : 0;
      state.s6EnlargeAnim += (targetEnlarge - state.s6EnlargeAnim) * 0.16;

      const arcRadiusX = Math.min(state.width * 0.62, 340);
      const arcDepthZ = 220;
      const baseCardW = Math.min(state.width * 0.45, 200);
      const baseCardH = baseCardW * (450 / 360);

      for (let i = 0; i < N; i++) {
        const card = ribbonCards[i];
        let delta = ((i - state.ribbonPos) % N + N) % N;
        if (delta > N * 0.5) delta -= N;

        const angle = delta * 0.44;
        const x3d = Math.sin(angle) * arcRadiusX;
        const z3d = (1 - Math.cos(angle)) * arcDepthZ;
        const persp = 620 / (620 + z3d);

        const yWave = Math.sin(delta * 0.65 + state.time * 1.1) * 10 - (1 - Math.cos(angle)) * 18;

        let sx = cx + x3d * persp;
        let sy = cy + yWave * persp;
        let scale = persp * Math.max(0.32, 1 - Math.abs(delta) * 0.11);
        let alpha = Math.max(0, 1 - Math.abs(delta) * 0.22);

        if (state.s6ExitingVortex) {
          const vp = state.s6VortexProgress;
          const spiralAngle = vp * Math.PI * 3.5 + i * 0.4;
          const shrink = 1 - vp;
          sx = cx + (sx - cx) * shrink * Math.cos(spiralAngle) - (sy - cy) * shrink * Math.sin(spiralAngle);
          sy = cy + (sx - cx) * shrink * Math.sin(spiralAngle) + (sy - cy) * shrink * Math.cos(spiralAngle);
          scale *= shrink;
          alpha *= shrink;
        }

        card.screenX = sx;
        card.screenY = sy;
        card.screenZ = z3d;
        card.drawW = baseCardW * scale;
        card.drawH = baseCardH * scale;
        card.alpha = alpha;
        card.isFront = Math.abs(delta) < 0.45;
      }

      const sortedCards = [...ribbonCards].sort((a, b) => b.screenZ - a.screenZ);

      for (let i = 0; i < sortedCards.length; i++) {
        const c = sortedCards[i];
        if (c.alpha <= 0.02 || c.drawW <= 2) continue;

        ctx.save();
        ctx.globalAlpha = c.alpha * (1 - state.s6EnlargeAnim * 0.72);
        ctx.translate(c.screenX, c.screenY);

        if (c.isFront && !state.s6ExitingVortex) {
          ctx.shadowColor = 'rgba(247, 227, 175, 0.68)';
          ctx.shadowBlur = 24;
        }

        ctx.drawImage(c.canvas, -c.drawW * 0.5, -c.drawH * 0.5, c.drawW, c.drawH);

        if (c.isFront && !state.s6ExitingVortex) {
          ctx.strokeStyle = 'rgba(247, 227, 175, 0.88)';
          ctx.lineWidth = 2;
          ctx.strokeRect(-c.drawW * 0.5, -c.drawH * 0.5, c.drawW, c.drawH);
        }

    function updateAndRenderParticles() {
      const cx = state.width * 0.5;
      const cy = state.height * 0.44;
      const arcRadiusX = Math.min(state.width * 0.62, 340);

      const easeZoom = 1 - Math.pow(1 - state.s3EnterProgress, 3);
      const cakeScale = (state.scene === 3 || state.scene === 4) ? (0.42 + (1.0 - 0.42) * easeZoom) : 1;

      for (let i = 0; i < particles.length; i++) {
        const p = particles[i];
        if (!p.active) continue;

        if (p.role === 'ribbonTrack' && state.scene === 6) {
          const angle = (p.trackFrac - 0.5) * Math.PI * 1.45 + state.ribbonPos * 0.12;
          const rx = Math.sin(angle) * arcRadiusX * 1.15;
          const ry = Math.cos(angle) * 42 + 65 + p.trackOffset;
          if (state.s6ExitingVortex) {
            const shrink = 1 - state.s6VortexProgress;
            p.tx = cx + rx * shrink;
            p.ty = cy + ry * shrink;
          } else {
            p.tx = cx + rx;
            p.ty = cy + ry;
          }
          p.x += (p.tx - p.x) * 0.16;
          p.y += (p.ty - p.y) * 0.16;
        } else if (p.role === 'ambient') {
          p.x += Math.cos(state.time * 0.6 + p.phase) * 0.28 + p.vx;
          p.y += Math.sin(state.time * 0.5 + p.phase) * 0.28 + p.vy;
          p.vx *= 0.96;
          p.vy *= 0.96;
          if (p.x < -10) p.x = state.width + 10;
          if (p.x > state.width + 10) p.x = -10;
          if (p.y < -10) p.y = state.height + 10;
          if (p.y > state.height + 10) p.y = -10;
        } else {
          let targetX = p.tx;
          let targetY = p.ty;
          if (state.scene === 3 || state.scene === 4) {
            targetX = cakeGeometry.cx + (p.tx - cakeGeometry.cx) * cakeScale;
            targetY = cakeGeometry.baseY + (p.ty - cakeGeometry.baseY) * cakeScale;
          }
          p.vx = (p.vx + (targetX - p.x) * 0.085) * 0.78;
          p.vy = (p.vy + (targetY - p.y) * 0.085) * 0.78;
          p.x += p.vx;
          p.y += p.vy;
        }

        const twinkle = 0.72 + 0.28 * Math.sin(state.time * 3.2 + p.phase);
        ctx.globalAlpha = Math.max(0.05, Math.min(1, p.alpha * twinkle));

        if (state.qualityTier >= 1 && p.role !== 'ambient') {
          const sprite = glowSprites[p.color];
          const s = p.size * 4.4;
          ctx.drawImage(sprite, p.x - s * 0.5, p.y - s * 0.5, s, s);
        } else {
          ctx.fillStyle = p.color;
          ctx.beginPath();
          ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
          ctx.fill();
        }
      }
      ctx.globalAlpha = 1;
    }

    /**
     * UPDATE & RENDER TRANSIENT FX PARTICLES
     */
    function updateAndRenderFxParticles() {
      for (let i = fxParticles.length - 1; i >= 0; i--) {
        const fp = fxParticles[i];
        fp.life -= fp.decay;
        if (fp.life <= 0) {
          fxParticles.splice(i, 1);
          continue;
        }

        fp.vx *= fp.drag;
        fp.vy = fp.vy * fp.drag + fp.gravity;
        if (fp.type === 'smoke') {
          fp.x += fp.vx + Math.sin(state.time * 3 + fp.phase) * 0.35;
          fp.size += 0.14;
        } else {
          fp.x += fp.vx;
          fp.y += fp.vy;
        }

        ctx.globalAlpha = Math.max(0, fp.life);
        if (fp.type === 'smoke') {
          const grad = ctx.createRadialGradient(fp.x, fp.y, 0, fp.x, fp.y, fp.size);
          grad.addColorStop(0, 'rgba(216, 180, 254, 0.35)');
          grad.addColorStop(1, 'rgba(216, 180, 254, 0)');
          ctx.fillStyle = grad;
          ctx.beginPath();
          ctx.arc(fp.x, fp.y, fp.size, 0, Math.PI * 2);
          ctx.fill();
        } else {
          const sprite = glowSprites[fp.color] || glowSprites['#FFFDF9'];
          const s = fp.size * 4.5 * fp.life;
          ctx.drawImage(sprite, fp.x - s * 0.5, fp.y - s * 0.5, s, s);
        }
      }
      ctx.globalAlpha = 1;
    }

    let pointerDown = false;
    let pointerStartX = 0;
    let pointerStartY = 0;
    let pointerLastX = 0;
    let pointerLastY = 0;
    let pointerMoved = 0;

    function handlePointerStart(x, y) {
      pointerDown = true;
      pointerStartX = x;
      pointerStartY = y;
      pointerLastX = x;
      pointerLastY = y;
      pointerMoved = 0;

      if (state.scene === 6 && state.s6GrabbedIdx === -1 && !state.s6ExitingVortex) {
        state.isDraggingRibbon = true;
        state.ribbonVel = 0;
      }
    }

    function handlePointerMove(x, y) {
      if (!pointerDown) return;
      const dx = x - pointerLastX;
      const dy = y - pointerLastY;
      pointerMoved += Math.hypot(dx, dy);
      pointerLastX = x;
      pointerLastY = y;

      // Scene 2.2: Swiping across screen erases "2"
      if (state.scene === 2 && state.s2Step === 1 && state.s2SubPhase === 'erasing' && pointerMoved > 35) {
        triggerEraseTwo();
      }

      // Scene 6: Drag horizontally to scroll Holographic Starlight Ribbon, or swipe enlarged card
      if (state.scene === 6 && !state.s6ExitingVortex) {
        if (state.s6GrabbedIdx !== -1) {
          state.s6ModalSwipeX = x - pointerStartX;
          state.s6ModalSwipeY = Math.max(0, y - pointerStartY);
        } else if (state.isDraggingRibbon) {
          const deltaSlots = -dx / Math.max(110, state.width * 0.28);
          state.ribbonPos += deltaSlots;
          state.ribbonVel = deltaSlots;
        }
      }
    }

    function handlePointerEnd(x, y) {
      if (!pointerDown) return;
      pointerDown = false;
      state.isDraggingRibbon = false;

      const isTap = pointerMoved < 14;
      const now = performance.now();

      if (state.scene === 2 && isTap) {
        advanceScene2();
        return;
      }

      if (state.scene === 3 && isTap && !state.candleLit) {
        igniteCandle();
        return;
      }

      if (state.scene === 4 && isTap && state.candleLit && !state.candleExtinguishedLocked) {
        state.s4TapTimestamps.push(now);
        state.s4TapTimestamps = state.s4TapTimestamps.filter((t) => now - t <= 1400);
        if (state.s4TapTimestamps.length >= 3) {
          extinguishCandle();
        }
        return;
      }

      if (state.scene === 5 && isTap) {
        if (!state.s5AllShown) {
          while (state.s5BubblesSpawned < BLESSING_LINES.length) {
            spawnBlessingBubble(state.s5BubblesSpawned);
            state.s5BubblesSpawned += 1;
          }
          state.s5AllShown = true;
          setPrompt('张开右手保持（或轻触屏幕），开启全息星河相册 ✋');
        } else {
          enterScene(6);
        }
        return;
      }

      if (state.scene === 6 && !state.s6ExitingVortex) {
        if (state.s6GrabbedIdx !== -1) {
          const totalDx = x - pointerStartX;
          const totalDy = y - pointerStartY;
          if (totalDy > 65) {
            triggerReleasePhoto();
          } else if (Math.abs(totalDx) > 45) {
            const N = ribbonCards.length;
            if (totalDx > 0) {
              state.s6GrabbedIdx = (state.s6GrabbedIdx - 1 + N) % N;
            } else {
              state.s6GrabbedIdx = (state.s6GrabbedIdx + 1) % N;
            }
            state.ribbonPos = state.s6GrabbedIdx;
          } else if (isTap) {
            const cx = state.width * 0.5;
            const cy = state.height * 0.44;
            const maxCardW = Math.min(state.width * 0.82, 340);
            const maxCardH = maxCardW * (450 / 360);
            if (Math.abs(x - cx) > maxCardW * 0.5 || Math.abs(y - cy) > maxCardH * 0.5) {
              triggerReleasePhoto();
            }
          }
          state.s6ModalSwipeX = 0;
          state.s6ModalSwipeY = 0;
        } else if (isTap) {
          const frontToBack = [...ribbonCards].sort((a, b) => a.screenZ - b.screenZ);
          for (let i = 0; i < frontToBack.length; i++) {
            const c = frontToBack[i];
            if (c.alpha < 0.25) continue;
            if (Math.abs(x - c.screenX) <= c.drawW * 0.54 && Math.abs(y - c.screenY) <= c.drawH * 0.54) {
              if (c.isFront) {
                triggerAirGrabPhoto(c.index);
              } else {
                const N = ribbonCards.length;
                let delta = ((c.index - state.ribbonPos) % N + N) % N;
                if (delta > N * 0.5) delta -= N;
                state.ribbonPos += delta;
                state.ribbonVel = 0;
              }
              break;
            }
          }
        }
      }
    }

    async function decodeAndCompressPhotoFile(file) {
      let usableBlob = file;
      const lowerName = (file.name || '').toLowerCase();
      const isHeic = lowerName.endsWith('.heic') || lowerName.endsWith('.heif') || file.type === 'image/heic' || file.type === 'image/heif';

      // Attempt native browser decode first (Safari/iOS handles HEIC natively)
      let bitmapOrImg = null;
      try {
        bitmapOrImg = await loadBlobAsImage(usableBlob);
      } catch (e) {
        if (isHeic) {
          if (typeof window.heic2any === 'undefined') {
            await loadExternalScript('https://cdn.jsdelivr.net/npm/heic2any@0.0.4/dist/heic2any.min.js', 8000);
          }
          if (typeof window.heic2any !== 'undefined') {
            const converted = await window.heic2any({
              blob: file,
              toType: 'image/jpeg',
              quality: 0.86
            });
            usableBlob = Array.isArray(converted) ? converted[0] : converted;
            bitmapOrImg = await loadBlobAsImage(usableBlob);
          } else {
            throw e;
          }
        } else {
          throw e;
        }
      }

      // Draw onto offscreen canvas with max dimension 920px for crisp mobile display & compact HTML size
      const maxDim = 920;
      let w = bitmapOrImg.width;
      let h = bitmapOrImg.height;
      if (w > maxDim || h > maxDim) {
        if (w >= h) {
          h = Math.round((h * maxDim) / w);
          w = maxDim;
        } else {
          w = Math.round((w * maxDim) / h);
          h = maxDim;
        }
      }

      const off = document.createElement('canvas');
      off.width = w;
      off.height = h;
      const octx = off.getContext('2d');
      octx.drawImage(bitmapOrImg, 0, 0, w, h);
      return off.toDataURL('image/jpeg', 0.84);
    }

    function loadBlobAsImage(blob) {
      return new Promise((resolve, reject) => {
        const objUrl = URL.createObjectURL(blob);
        const img = new Image();
        img.onload = () => {
          URL.revokeObjectURL(objUrl);
          resolve(img);
        };
        img.onerror = (err) => {
          URL.revokeObjectURL(objUrl);
          reject(err);
        };
        img.src = objUrl;
      });
    }

      const now = performance.now();
      if (now - lastStartActionTime < 500) return;
      lastStartActionTime = now;

      // 1. FIRST LINE: Immediately & synchronously switch into Scene 1 (5->4->3->2->1)
      enterScene(1);

      // 2. Asynchronously start background audio (with Web Audio music-box fallback)
      state.audioStarted = true;
      const audioBtn = document.getElementById('audio-toggle-btn');
      audioBtn.classList.remove('hidden');

      if (bgmEl) {
        const playPromise = bgmEl.play();
        if (playPromise && typeof playPromise.catch === 'function') {
          playPromise.catch(() => {
            musicBoxFallback.start();
          });
        }
      } else {
        musicBoxFallback.start();
      }

      // 3. Asynchronously initialize local camera & MediaPipe in the background (non-blocking)
      initCameraAndVisionAsync();
    }

    startBtn.addEventListener('touchend', handleStartExperience, { passive: false });
    startBtn.addEventListener('click', handleStartExperience);

    document.getElementById('audio-toggle-btn').addEventListener('click', (e) => {
      e.stopPropagation();
      state.isMuted = !state.isMuted;
      if (bgmEl) bgmEl.muted = state.isMuted;
      document.getElementById('icon-sound-on').classList.toggle('hidden', state.isMuted);
      document.getElementById('icon-sound-off').classList.toggle('hidden', !state.isMuted);
      if (!state.isMuted && bgmEl && bgmEl.paused && !musicBoxFallback.active) {
        bgmEl.play().catch(() => musicBoxFallback.start());
      }
    });

    document.querySelectorAll('.s1-tap-btn').forEach((btn) => {
      const onTapDigit = (e) => {
        if (e && e.cancelable) e.preventDefault();
        if (state.scene !== 1) return;
        const tappedNum = parseInt(btn.getAttribute('data-tap-num'), 10);
        if (tappedNum === state.s1Target) {
          const rect = btn.getBoundingClientRect();
          collectScene1Number(rect.left + rect.width * 0.5, rect.top + rect.height * 0.5);
        }
      };
      btn.addEventListener('touchend', onTapDigit, { passive: false });
      btn.addEventListener('click', onTapDigit);
    });

    const startHoldBlow = (e) => {
      if (e) e.stopPropagation();
      state.s4IsHoldingBlow = true;
    };
    const endHoldBlow = (e) => {
      if (e) e.stopPropagation();
      state.s4IsHoldingBlow = false;
    };
    blowHoldBtn.addEventListener('mousedown', startHoldBlow);
    blowHoldBtn.addEventListener('mouseup', endHoldBlow);
    blowHoldBtn.addEventListener('mouseleave', endHoldBlow);
    blowHoldBtn.addEventListener('touchstart', startHoldBlow, { passive: true });
    blowHoldBtn.addEventListener('touchend', endHoldBlow, { passive: true });

    document.getElementById('s6-exit-finale-btn').addEventListener('click', (e) => {
      e.stopPropagation();
      if (state.scene === 6) {
        triggerVortexExitToFinale();
      }
    });

    const replayBtn = document.getElementById('replay-btn');
    const handleReplay = (e) => {
      if (e && e.cancelable) e.preventDefault();
      enterScene(0);
    };
    replayBtn.addEventListener('touchend', handleReplay, { passive: false });
    replayBtn.addEventListener('click', handleReplay);

    /**
     * ============================================================================
     * INITIALIZE STAGE ON DOM CONTENT LOADED / WINDOW LOAD
     * ============================================================================

    function bootStage() {
      resizeCanvas();
      initMainParticles();
      enterScene(0);
      requestAnimationFrame(updateAndRender);
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', bootStage);
    } else {
      bootStage();
    }

      <script>
        (function() {
          // The root element's background fills the whole page. As in CSS,
          // the body's background is used when the root element has none.
          function getPageBackgroundColor() {
            var docStyle = window.getComputedStyle(document.documentElement);
            var color = docStyle.backgroundColor;
            if ((color === 'rgba(0, 0, 0, 0)' || color === 'transparent') &&
                docStyle.backgroundImage === 'none' && document.body) {
              color = window.getComputedStyle(document.body).backgroundColor;
            }
            return color || '';
          }

          var lastReportedColor = '';
          function reportBackgroundColor() {
            try {
              var color = getPageBackgroundColor();
              if (color === lastReportedColor) return;
              lastReportedColor = color;
              window.parent.postMessage({
                type: 'ATARI_CODE_EMBED_BACKGROUND_COLOR',
                backgroundColor: color
              }, '*');
            } catch (e) {}
          }

          // The page background usually changes through an attribute on html
          // or body (for example a theme class), a style sheet in head, or the
          // system color scheme.
          if (typeof MutationObserver !== 'undefined') {
            var observer = new MutationObserver(reportBackgroundColor);
            observer.observe(document.documentElement, {attributes: true});
            if (document.body) {
              observer.observe(document.body, {attributes: true});
            }
            if (document.head) {
              observer.observe(document.head, {
                attributes: true,
                childList: true,
                subtree: true,
                characterData: true
              });
            }
          }
          var colorSchemeQuery =
              window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)');
          if (colorSchemeQuery && colorSchemeQuery.addEventListener) {
            colorSchemeQuery.addEventListener('change', reportBackgroundColor);
          }
          window.addEventListener('load', reportBackgroundColor);
          window.addEventListener('transitionend', reportBackgroundColor);
          window.addEventListener('animationend', reportBackgroundColor);
          reportBackgroundColor();
        })();
      </script>
